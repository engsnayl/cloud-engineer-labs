# =============================================================================
# PYTHON DEPENDENCIES — Libraries our backend needs to run
# =============================================================================
# This file tells pip (Python's package manager) which libraries to install.
# In the Dockerfile, we run "pip install -r requirements.txt" to install these.
#
# Pinning versions (e.g., Flask==3.0.0) is best practice for production because
# it prevents surprise breakages when a new version is released. For this
# learning project, we use >= (minimum version) for simplicity.
# =============================================================================

# Flask — The web framework that handles HTTP requests and routing.
# It's what lets us define @app.route("/api/health") and have it respond to HTTP calls.
Flask>=3.0.0

# psycopg2-binary — PostgreSQL database driver for Python.
# The "-binary" variant comes pre-compiled, so you don't need PostgreSQL
# development headers installed on your system to build it.
# In production, you'd use plain "psycopg2" and compile from source for
# maximum compatibility, but the binary version is fine for learning.
psycopg2-binary>=2.9.9

# gunicorn — A production-grade WSGI HTTP server for Python.
# Flask's built-in server (app.run()) is only meant for development.
# Gunicorn runs multiple worker processes to handle concurrent requests,
# which is what you need in a real deployment.
gunicorn>=21.2.0
